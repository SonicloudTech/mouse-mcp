cmake_minimum_required(VERSION 3.20)
project(umouse_sdk_host VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "First implementation supports Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(UMCP_BUILD_TESTS "Build isolated fake plugins and tests" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(umcp_json STATIC third_party/yyjson/yyjson.c)
target_include_directories(umcp_json PUBLIC third_party/yyjson)
add_library(umcp_common INTERFACE)
target_include_directories(umcp_common INTERFACE include src)
target_link_libraries(umcp_common INTERFACE umcp_json bcrypt advapi32 user32)
if(MSVC)
  target_compile_options(umcp_common INTERFACE /W4 /utf-8 /EHsc)
endif()
add_executable(umouse-sdk-host src/main.cpp)
target_link_libraries(umouse-sdk-host PRIVATE umcp_common iphlpapi)
foreach(component umouse typer asr screen)
  add_library(umcp_${component} SHARED src/${component}_plugin.cpp)
  target_link_libraries(umcp_${component} PRIVATE umcp_common)
endforeach()
set(UMCP_UMOUSE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../include" CACHE PATH "Public SDK C header directory; SDK implementation is never linked")
if(NOT EXISTS "${UMCP_UMOUSE_INCLUDE}/umouse_sdk.h")
  message(FATAL_ERROR "Set UMCP_UMOUSE_INCLUDE to the public SDK header directory. No SDK source tree is required.")
endif()
target_include_directories(umcp_umouse PRIVATE "${UMCP_UMOUSE_INCLUDE}")
if(UMCP_BUILD_TESTS)
  add_library(umcp_fake SHARED test/fake_plugin.cpp)
  target_link_libraries(umcp_fake PRIVATE umcp_common)
  add_library(umcp_bad_abi SHARED test/fake_plugin.cpp)
  target_compile_definitions(umcp_bad_abi PRIVATE UMCP_BAD_ABI=1)
  target_link_libraries(umcp_bad_abi PRIVATE umcp_common)
  foreach(vendor umcp_test_vendor umcp_test_vendor_legacy_screen)
    add_library(${vendor} SHARED test/vendor_stub.cpp)
    target_include_directories(${vendor} PRIVATE "${UMCP_UMOUSE_INCLUDE}")
    target_compile_definitions(${vendor} PRIVATE UMSDK_EXPORTS)
    target_link_libraries(${vendor} PRIVATE umcp_common)
  endforeach()
  target_compile_definitions(umcp_test_vendor_legacy_screen PRIVATE UMCP_NO_SCREEN_WAIT)
  add_executable(umcp-adapter-contract test/adapter_contract.cpp)
  target_link_libraries(umcp-adapter-contract PRIVATE umcp_common)
  add_executable(umcp-hardware-contract test/hardware_contract.cpp)
  target_link_libraries(umcp-hardware-contract PRIVATE umcp_common)
  add_executable(umcp-raw-feedback-contract test/raw_feedback_contract.cpp)
  target_link_libraries(umcp-raw-feedback-contract PRIVATE umcp_common)
  add_executable(umcp-sequence-contract test/sequence_contract.cpp)
  target_link_libraries(umcp-sequence-contract PRIVATE umcp_common)
  add_executable(umcp-input-quarantine-contract test/input_quarantine_contract.cpp)
  target_link_libraries(umcp-input-quarantine-contract PRIVATE umcp_common)
  add_executable(umcp-diagnostics-contract test/diagnostics_contract.cpp)
  target_link_libraries(umcp-diagnostics-contract PRIVATE umcp_common)
  add_executable(umcp-job-parent test/job_parent.cpp)
  target_link_libraries(umcp-job-parent PRIVATE umcp_common)
endif()
